**Source code:** [`Task2_JDBC/src/main/java/PrimeInsert.java`](Task2_JDBC/src/main/java/PrimeInsert.java)

(import java.sql.*;
import java.util.Scanner;

public class PrimeInsert {

    // Connection settings (change password to yours)
    private static final String URL  = "jdbc:mysql://localhost:3306/lab_db?useSSL=false&serverTimezone=UTC";
    private static final String USER = "root";
    private static final String PASS = "sql@123";

    // Table name format: nickname_name_prime  (replace with yours)
    private static final String TABLE = "Mashrafi_name_prime";

    // Checks whether a number is prime using trial division up to sqrt(num)
    static boolean isPrime(int num) {
        if (num < 2) return false;
        if (num % 2 == 0) return num == 2;
        for (int i = 3; (long) i * i <= num; i += 2) {
            if (num % i == 0) return false;
        }
        return true;
    }

    // Returns the n-th prime number (1st prime = 2)
    static int nthPrime(int n) {
        int count = 0, candidate = 1;
        while (count < n) {
            candidate++;
            if (isPrime(candidate)) count++;
        }
        return candidate;
    }

    public static void main(String[] args) {
        Scanner sc = new Scanner(System.in);
        System.out.print("Enter n: ");
        int n = sc.nextInt();
        if (n < 1) {
            System.out.println("n must be at least 1.");
            return;
        }

        int prime = nthPrime(n);
        System.out.println("The " + n + "-th prime number is: " + prime);

        // try-with-resources closes the connection automatically
        try (Connection con = DriverManager.getConnection(URL, USER, PASS)) {
            System.out.println("Connected to MySQL successfully.");

            // 1. Create the table if it does not exist
            try (Statement st = con.createStatement()) {
                st.executeUpdate("CREATE TABLE IF NOT EXISTS " + TABLE + " (" +
                        "id INT AUTO_INCREMENT PRIMARY KEY, " +
                        "n INT NOT NULL, " +
                        "prime_value INT NOT NULL)");
            }

            // 2. Insert the result (PreparedStatement prevents SQL injection)
            String insertSql = "INSERT INTO " + TABLE + " (n, prime_value) VALUES (?, ?)";
            try (PreparedStatement ps = con.prepareStatement(insertSql)) {
                ps.setInt(1, n);
                ps.setInt(2, prime);
                ps.executeUpdate();
                System.out.println("Record inserted.");
            }

            // 3. Retrieve and display the stored records
            try (Statement st = con.createStatement();
                 ResultSet rs = st.executeQuery("SELECT * FROM " + TABLE)) {
                System.out.println("\nid | n | prime_value");
                while (rs.next()) {
                    System.out.println(rs.getInt("id") + "  | " +
                            rs.getInt("n") + " | " + rs.getInt("prime_value"));
                }
            }
        } catch (SQLException e) {
            System.out.println("Database error: " + e.getMessage());
        }
    }
}
